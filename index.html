استعلام SQL الجاهز للإنشاء

انسخ الكود والصقه في SQL Editor في Supabase

-- 1. إنشاء جدول المهام
create table if not exists todos (
  id uuid default gen_random_uuid() primary key,
  title text not null,
  due_date timestamptz,
  priority text default 'medium',
  category text default 'عام',
  notes text,
  completed boolean default false,
  created_at timestamptz default now()
);

-- 2. السماح بالقراءة والكتابة للجميع (Public Access)
alter table todos enable row level security;
drop policy if exists "Public Access" on todos;
create policy "Public Access" on todos for all using (true) with check (true);

-- 3. تفعيل المزامنة اللحظية (Realtime)
alter publication supabase_realtime add table todos;

💡 طريقة الاستخدام: افتح مشروعك على Supabase 👈 ثم SQL Editor 👈 أنشئ New Query 👈 الصق الكود واضغط Run.
